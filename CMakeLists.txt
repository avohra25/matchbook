cmake_minimum_required(VERSION 3.20)
project(matchbook VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(MATCHBOOK_SANITIZE "Build with AddressSanitizer and UBSan" OFF)

file(GLOB MATCHBOOK_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(matchbook ${MATCHBOOK_SOURCES})
target_include_directories(matchbook PUBLIC include)
target_compile_options(matchbook PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wno-missing-field-initializers -Werror)
if(MATCHBOOK_SANITIZE)
  target_compile_options(matchbook PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(matchbook PUBLIC -fsanitize=address,undefined)
endif()

enable_testing()
file(GLOB MATCHBOOK_TESTS CONFIGURE_DEPENDS tests/test_*.cpp)
foreach(test_src ${MATCHBOOK_TESTS})
  get_filename_component(test_name ${test_src} NAME_WE)
  add_executable(${test_name} ${test_src})
  target_link_libraries(${test_name} PRIVATE matchbook)
  target_compile_options(${test_name} PRIVATE -Wall -Wextra -Wno-missing-field-initializers -Werror)
  add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
